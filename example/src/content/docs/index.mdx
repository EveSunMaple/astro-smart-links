## 安装

```bash
npm install astro-smart-links
```

## 使用

```js
// astro.config.mjs
import { defineConfig } from 'astro/config';
import { smartLinks } from 'astro-smart-links';

export default defineConfig({
  integrations: [smartLinks()],
});
```
